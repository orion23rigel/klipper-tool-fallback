    def cmd_DEFINE_TOOL_BACKUP(self, gcmd):
        # 1. Guard: notification operation (D-13: consistent with existing backup/mapping commands)
        self._guard_notification_operation(gcmd.error)
        # 2. Guard: workflow operation (D-10: same guard used by cmd_REMAP_TOOL and cmd_SET_TOOL_BACKUPS)
        self._guard_workflow_operation(gcmd.error, "tool backup definition")
        # 3. Validate logical tool is configured
        logical_tool = self._require_configured_tool(gcmd, "LOGICAL")
        # 4. Validate backup tool is configured (D-08: reuse _require_configured_tool or inline check against self.config.tools)
        backup_tool = self._require_configured_tool(gcmd, "BACKUP")
        # 5. Validate self-reference (logical == backup) (D-09: reject self-references consistent with Phase 8 D-14)
        if logical_tool == backup_tool:
            raise gcmd.error(
                "Tool %s cannot be its own backup" % (logical_tool,))
        # 6. Mutate state
        candidate = self.state.with_user_defined_backup(logical_tool, backup_tool)
        # 7. No-op detection
        if candidate is self.state:
            gcmd.respond_info(
                "Tool %s backup unchanged: %s -> %s (no write)" %
                (logical_tool,
                 self.state.user_defined_backups.get(logical_tool, "(none)"),
                 backup_tool))
            return
        # 8. Persist
        try:
            self._persist_state(candidate)
        except OSError as error:
            raise gcmd.error(
                "Unable to persist tool fallback backup mapping: %s" % (error,))
        # 9. Success response
        gcmd.respond_info(
            "Tool %s user backup set to %s" % (logical_tool, backup_tool))
    ```

    **Parameter names:** Use uppercase `LOGICAL` and `BACKUP` per D-04 (Klipper convention uses uppercase parameter names).

    **Error messages:** Use `gcmd.error()` for failures (D-17: validation errors use gcmd.error with descriptive messages matching Klipper conventions). Use `gcmd.respond_info()` for success (D-18: success messages use gcmd.respond_info with human-readable output showing what was changed).

    **Do NOT:** Add workflow guard for UNDEFINE here — that's in Task 2. Do NOT add SHOW_TOOL_BACKUPS here — that's a separate plan.

    def cmd_UNDEFINE_TOOL_BACKUP(self, gcmd):
        # 1. Guard: notification operation
        self._guard_notification_operation(gcmd.error)
        # 2. Guard: workflow operation (D-11 — consistent with existing mutation pattern)
        self._guard_workflow_operation(gcmd.error, "tool backup undefinition")
        # 3. Validate logical tool is configured (D-07: _require_configured_tool ensures it's a configured tool)
        logical_tool = self._require_configured_tool(gcmd, "TOOL")
        # 4. Validate mapping exists (D-06: validate the mapping exists before removing)
        if logical_tool not in self.state.user_defined_backups:
            raise gcmd.error(
                "No user-defined backup for tool %s" % (logical_tool,))
        # 5. Mutate state (None = remove)
        candidate = self.state.with_user_defined_backup(logical_tool, None)
        # 6. Persist
        try:
            self._persist_state(candidate)
        except OSError as error:
            raise gcmd.error(
                "Unable to persist tool fallback backup mapping: %s" % (error,))
        # 7. Success response
        gcmd.respond_info(
            "Tool %s user backup removed" % (logical_tool,))
    ```
